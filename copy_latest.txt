S_DIR="/path/to/source"
D_DIR="/path/to/dest"

# 1. Очистить папку D_DIR (все файлы и поддиректории)
rm -rf "$D_DIR"/*

# 2. Найти самый свежий файл в S_DIR по имени (kk_YYYYMMDD.gz)
LATEST_FILE=$(ls -1 "$S_DIR"/kk_*.gz 2>/dev/null | sort -t_ -k2 -r | head -n 1)

# 3. Проверка что файл найден
if [[ -n "$LATEST_FILE" ]]; then
    cp "$LATEST_FILE" "$D_DIR"/
    echo "Скопирован файл: $LATEST_FILE → $D_DIR/"
else
    echo "Файлы kk_YYYYMMDD.gz не найдены в $S_DIR"
fi
