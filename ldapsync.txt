#!/bin/bash

# -------------------------------
# Настройки
# -------------------------------
CONFIG_FILE="/etc/ldap_sync/realms.json"   # путь к JSON с настройками
LOG_FILE="/var/log/ldap_sync.log"          # лог файл

# -------------------------------
# Проверка аргумента и выбор типа синхронизации
# -------------------------------
case "$1" in
    full)
        SYNC_TYPE="triggerFullSync"
        SYNC_NAME="полная"
        ;;
    changed)
        SYNC_TYPE="triggerChangedUsersSync"
        SYNC_NAME="только изменённые пользователи"
        ;;
    *)
        echo "Использование: $0 <full|changed>"
        exit 1
        ;;
esac

# -------------------------------
# Функция логирования
# -------------------------------
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a "$LOG_FILE"
}

# -------------------------------
# Получение токена доступа для Admin API
# -------------------------------
get_token() {
    local BASE_URL=$1       # базовый URL Keycloak
    local REALM_NAME=$2     # имя realm (из JSON)
    local CLIENT_ID=$3
    local CLIENT_SECRET=$4

    echo "Получаем токен для realm '$REALM_NAME'..."
    TOKEN=$(curl -s -X POST "$BASE_URL/realms/$REALM_NAME/protocol/openid-connect/token" \
        -H "Content-Type: application/x-www-form-urlencoded" \
        -d "grant_type=client_credentials" \
        -d "client_id=$CLIENT_ID" \
        -d "client_secret=$CLIENT_SECRET" | jq -r '.access_token')

    if [[ -z "$TOKEN" || "$TOKEN" == "null" ]]; then
        echo "Ошибка: не удалось получить токен для realm '$REALM_NAME' с client_id='$CLIENT_ID'"
        return 1
    else
        echo "Токен успешно получен"
    fi

    echo "$TOKEN"
}

# -------------------------------
# Функция синхронизации LDAP для одного realm
# -------------------------------
sync_ldap() {
    local BASE_URL=$1
    local REALM_NAME=$2
    local TOKEN=$3
    local TYPE=$4
    local NAME=$5

    echo "Получаем список LDAP-провайдеров для realm '$REALM_NAME'..."
    LDAP_PROVIDERS=$(curl -s -X GET "$BASE_URL/admin/realms/$REALM_NAME/user-storage" \
        -H "Authorization: Bearer $TOKEN" \
        -H "Content-Type: application/json" | jq -r '.[].id')

    if [[ -z "$LDAP_PROVIDERS" ]]; then
        echo "Внимание: LDAP-провайдеров не найдено для realm '$REALM_NAME'"
        log "Внимание: LDAP-провайдеров не найдено для realm '$REALM_NAME'"
        return
    fi

    for PROVIDER_ID in $LDAP_PROVIDERS; do
        echo "Запуск $NAME синхронизации LDAP для провайдера '$PROVIDER_ID'..."
        log "Запуск $NAME синхронизации LDAP для realm '$REALM_NAME', провайдер '$PROVIDER_ID'"

        RESPONSE=$(curl -s -o /dev/null -w "%{http_code}" -X POST \
            "$BASE_URL/admin/realms/$REALM_NAME/user-storage/$PROVIDER_ID/sync?action=$TYPE" \
            -H "Authorization: Bearer $TOKEN" \
            -H "Content-Type: application/json")

        if [[ "$RESPONSE" == "204" ]]; then
            echo "$NAME синхронизация LDAP успешно завершена для провайдера '$PROVIDER_ID'"
            log "$NAME синхронизация LDAP успешно завершена для провайдера '$PROVIDER_ID'"
        else
            echo "$NAME синхронизация LDAP НЕ удалась для провайдера '$PROVIDER_ID', HTTP код: $RESPONSE"
            log "$NAME синхронизация LDAP НЕ удалась для провайдера '$PROVIDER_ID', HTTP код: $RESPONSE"
        fi
    done
}

# -------------------------------
# Загружаем конфигурацию
# -------------------------------
CONFIG=$(cat "$CONFIG_FILE")

# -------------------------------
# Основной цикл по realms
# -------------------------------
REALM_COUNT=$(echo "$CONFIG" | jq '.realms | length')
echo "Всего realms в конфиге: $REALM_COUNT"

for i in $(seq 0 $((REALM_COUNT - 1))); do
    REALM_URL=$(echo "$CONFIG" | jq -r ".realms[$i].realm_URL")
    REALM_NAME=$(echo "$CONFIG" | jq -r ".realms[$i].realm_name")
    CLIENT_ID=$(echo "$CONFIG" | jq -r ".realms[$i].client_id")
    CLIENT_SECRET=$(echo "$CONFIG" | jq -r ".realms[$i].client_secret")

    echo "Обработка realm '$REALM_NAME' по адресу '$REALM_URL'..."

    # Получаем токен (теперь realm берём из JSON!)
    TOKEN=$(get_token "$REALM_URL" "$REALM_NAME" "$CLIENT_ID" "$CLIENT_SECRET")
    if [[ $? -ne 0 ]]; then
        log "НЕ удалось получить токен для realm '$REALM_NAME'"
        continue
    fi

    # Синхронизация LDAP
    sync_ldap "$REALM_URL" "$REALM_NAME" "$TOKEN" "$SYNC_TYPE" "$SYNC_NAME"
done

log "Синхронизация LDAP завершена для всех realms"
echo "Синхронизация завершена. Проверяйте лог $LOG_FILE"


curl -v -X POST "https://keycloak/auth/realms/Group_test/protocol/openid-connect/token" \
  -H "Content-Type: application/x-www-form-urlencoded" \
  -d "grant_type=client_credentials" \
  -d "client_id=K_Admin" \
  -d "client_secret=123456"
