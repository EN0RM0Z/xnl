#!/bin/bash

# Скрипт синхронизации LDAP через Keycloak Admin API

CONFIG_FILE="config.json"

# Загружаем конфигурацию
if [[ ! -f "$CONFIG_FILE" ]]; then
    echo "Файл конфигурации $CONFIG_FILE не найден"
    exit 1
fi

CONFIG=$(cat "$CONFIG_FILE")

REALM_URL=$(echo "$CONFIG" | jq -r '.realm_URL')
REALM_NAME=$(echo "$CONFIG" | jq -r '.realm_name')
CLIENT_ID=$(echo "$CONFIG" | jq -r '.client_id')
CLIENT_SECRET=$(echo "$CONFIG" | jq -r '.client_secret')
LDAP_ID=$(echo "$CONFIG" | jq -r '.ldap_id')

# Функция получения токена
get_token() {
    local BASE_URL=$1
    local REALM_NAME=$2
    local CLIENT_ID=$3
    local CLIENT_SECRET=$4

    >&2 echo "----- Получение токена -----"
    >&2 echo "BASE_URL     = $BASE_URL"
    >&2 echo "REALM_NAME   = $REALM_NAME"
    >&2 echo "CLIENT_ID    = $CLIENT_ID"
    >&2 echo "CLIENT_SECRET= $CLIENT_SECRET"

    local URL="${BASE_URL}realms/${REALM_NAME}/protocol/openid-connect/token"
    >&2 echo "Запрос к URL: $URL"

    RESPONSE=$(curl -s -w "\n--- http_code=%{http_code} ---\n" -X POST "$URL" \
        -H "Content-Type: application/x-www-form-urlencoded" \
        -d "grant_type=client_credentials" \
        -d "client_id=${CLIENT_ID}" \
        -d "client_secret=${CLIENT_SECRET}")

    >&2 echo "Ответ от сервера:"
    >&2 echo "$RESPONSE"
    >&2 echo "----------------------------"

    TOKEN=$(echo "$RESPONSE" | jq -r '.access_token' 2>/dev/null)

    if [[ -z "$TOKEN" || "$TOKEN" == "null" ]]; then
        >&2 echo "Ошибка: токен не получен"
        return 1
    else
        echo "$TOKEN"
    fi
}

# Функция синхронизации LDAP
sync_ldap() {
    local BASE_URL=$1
    local REALM_NAME=$2
    local LDAP_ID=$3
    local TOKEN=$4

    >&2 echo "----- Синхронизация LDAP -----"
    >&2 echo "BASE_URL   = $BASE_URL"
    >&2 echo "REALM_NAME = $REALM_NAME"
    >&2 echo "LDAP_ID    = $LDAP_ID"

    local SYNC_URL="${BASE_URL}admin/realms/${REALM_NAME}/user-storage/${LDAP_ID}/sync"

    >&2 echo "Выполняется запрос к: $SYNC_URL"

    RESPONSE=$(curl -s -w "\n--- http_code=%{http_code} ---\n" -X POST "$SYNC_URL" \
        -H "Authorization: Bearer $TOKEN" \
        -H "Content-Type: application/json" \
        -d '{"action":"triggerFullSync"}')

    >&2 echo "Ответ от сервера:"
    >&2 echo "$RESPONSE"
    >&2 echo "-------------------------------"
}

# Основной процесс
echo "Запускаем получение токена..."
TOKEN=$(get_token "$REALM_URL" "$REALM_NAME" "$CLIENT_ID" "$CLIENT_SECRET")

if [[ -z "$TOKEN" ]]; then
    echo "Не удалось получить токен для realm '$REALM_NAME'"
    exit 1
fi

echo "Запускаем синхронизацию LDAP..."
sync_ldap "$REALM_URL" "$REALM_NAME" "$LDAP_ID" "$TOKEN"
