#!/bin/bash
set -euo pipefail

MAX_BACKUPS=7
TARGET_DIR="/var/backups/keycloak"
LOG_FILE="/var/log/backup_rotate.log"

echo "[INFO] Начало ротации каталогов в $TARGET_DIR" | tee -a "$LOG_FILE"

# 1. Удаляем самый старый каталог
if [ -d "${TARGET_DIR}.${MAX_BACKUPS}" ]; then
    echo "[INFO] Удаляю ${TARGET_DIR}.${MAX_BACKUPS}" | tee -a "$LOG_FILE"
    rm -rf "${TARGET_DIR}.${MAX_BACKUPS}"
fi

# 2. Сдвигаем каталоги с конца
for ((i=MAX_BACKUPS-1; i>=1; i--)); do
    if [ -d "${TARGET_DIR}.${i}" ]; then
        echo "[INFO] Переименовываю ${TARGET_DIR}.${i} → ${TARGET_DIR}.$((i+1))" | tee -a "$LOG_FILE"
        mv "${TARGET_DIR}.${i}" "${TARGET_DIR}.$((i+1))"
    fi
done

# 3. Текущий TARGET_DIR → TARGET_DIR.1
if [ -d "$TARGET_DIR" ]; then
    echo "[INFO] Переименовываю $TARGET_DIR → ${TARGET_DIR}.1" | tee -a "$LOG_FILE"
    mv "$TARGET_DIR" "${TARGET_DIR}.1"
fi

echo "[INFO] Ротация завершена" | tee -a "$LOG_FILE"
