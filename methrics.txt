#!/bin/bash
# keycloak_http_metrics.sh
# Подсчёт HTTP-кодов 200 и 50x в access.log Nginx (Keycloak)
# Оптимизировано для Zabbix: keycloak.http[200,5] или keycloak.http[50x]

set -euo pipefail

# ------------------------
# Конфигурация
# ------------------------
LOG_FILE="${LOG_FILE:-/var/log/nginx/keycloak.access.log}"
MAX_LINES="${MAX_LINES:-50000}"

CODE="$1"
MINUTES="${2:-5}"

# Обработка опционального второго параметра
if [[ -z "$MINUTES" ]] || ! [[ "$MINUTES" =~ ^[0-9]+$ ]] || [ "$MINUTES" -lt 1 ]; then
    MINUTES=5
fi

if [ "$MINUTES" -gt 1440 ]; then
    MINUTES=1440
fi

# ------------------------
# Валидация параметров
# ------------------------
if [[ "$CODE" != "200" && "$CODE" != "50x" ]]; then
    echo "ERROR: Code must be 200 or 50x" >&2
    exit 1
fi

# Быстрый выход если файл недоступен
if [[ ! -r "$LOG_FILE" ]] || [[ ! -s "$LOG_FILE" ]]; then
    echo "0"
    exit 0
fi

# ------------------------
# Время начала окна в epoch
# ------------------------
SINCE_EPOCH=$(date -d "-${MINUTES} minutes" +%s)

# ------------------------
# Функция для преобразования даты Nginx в epoch
# ------------------------
to_epoch_awk='
function to_epoch(str,    d, t_h, t_m, t_s, month_str, m, tz, sign, h, min, off, timestamp) {
    gsub(/^\[|\]$/,"",str)
    split(str, dt, ":")
    split(dt[1], d, "/")
    t_h = dt[2]; t_m = dt[3]; t_s = dt[4]
    month_str = "JanFebMarAprMayJunJulAugSepOctNovDec"
    m = int((index(month_str, d[2])-1)/3)+1
    if (m<1) return 0
    tz = dt[5]
    sign = substr(tz,1,1)
    h = substr(tz,2,2)+0
    min = substr(tz,4,2)+0
    off = (h*3600 + min*60)
    if (sign=="-") off = -off
    timestamp = mktime(d[3]" "m" "d[1]" "t_h" "t_m" "t_s) - off
    return (timestamp>0) ? timestamp : 0
}'

# ------------------------
# Подсчёт через awk
# ------------------------
COUNT=$(tail -n "$MAX_LINES" "$LOG_FILE" | awk -v since="$SINCE_EPOCH" -v code="$CODE" "$to_epoch_awk" '
{
    if (NF<9) next
    log_epoch = to_epoch($4 " " $5)
    if (log_epoch >= since) {
        # Разбор по кавычкам, чтобы корректно взять HTTP-код
        split($0,a,"\"")
        split(a[3],b," ")
        code_num = b[2]+0
        if ((code=="200" && code_num==200) || (code=="50x" && code_num>=500 && code_num<600)) count++
    }
}
END { print count+0 }')

echo "${COUNT:-0}"

function to_epoch(str,    datetime, d, t, m, month, tz, sign, h, min, off) {
    # Удаляем квадратные скобки
    gsub(/^\[|\]$/, "", str)
    
    # Разделяем дату/время и временную зону
    split(str, parts, " ")
    datetime = parts[1]
    tz = parts[2]
    
    # Парсим основную дату и время
    split(datetime, dt, ":")
    split(dt[1], d, "/")
    
    # Преобразуем месяц
    month_map["Jan"] = 1; month_map["Feb"] = 2; month_map["Mar"] = 3;
    month_map["Apr"] = 4; month_map["May"] = 5; month_map["Jun"] = 6;
    month_map["Jul"] = 7; month_map["Aug"] = 8; month_map["Sep"] = 9;
