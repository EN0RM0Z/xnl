#!/bin/bash
# keycloak_http_metrics.sh
# Подсчёт HTTP-кодов 200 и 50x в access.log Nginx (Keycloak)
# Устойчивый разбор формата access.log

set -euo pipefail

# ------------------------
# Конфигурация
# ------------------------
LOG_FILE="${LOG_FILE:-/var/log/nginx/keycloak.access.log}"
MAX_LINES="${MAX_LINES:-50000}"

CODE="${1:-200}"
MINUTES="${2:-5}"

# ------------------------
# Валидация параметров
# ------------------------
if ! [[ "$MINUTES" =~ ^[0-9]+$ ]] || [ "$MINUTES" -lt 1 ]; then
    echo "WARNING: MINUTES set to default 5" >&2
    MINUTES=5
elif [ "$MINUTES" -gt 1440 ]; then
    echo "WARNING: MINUTES capped at 1440 (24 hours)" >&2
    MINUTES=1440
fi

if [[ "$CODE" != "200" && "$CODE" != "50x" ]]; then
    echo "ERROR: Code must be 200 or 50x" >&2
    exit 1
fi

if [[ ! -f "$LOG_FILE" ]]; then
    echo "0"; exit 0
fi
if [[ ! -r "$LOG_FILE" ]]; then
    echo "ERROR: Cannot read log file: $LOG_FILE" >&2
    echo "0"; exit 0
fi
if [[ ! -s "$LOG_FILE" ]]; then
    echo "0"; exit 0
fi

# ------------------------
# Время начала окна
# ------------------------
SINCE_EPOCH=$(date -d "-${MINUTES} minutes" +%s 2>/dev/null || echo "0")
if [[ "$SINCE_EPOCH" == "0" ]]; then
    if command -v date >/dev/null 2>&1; then
        SINCE_EPOCH=$(( $(date +%s) - MINUTES * 60 ))
    else
        echo "ERROR: Cannot calculate timestamp" >&2
        echo "0"; exit 0
    fi
fi

# ------------------------
# Основная обработка через awk
# ------------------------
COUNT=$(LC_ALL=C tail -n "$MAX_LINES" "$LOG_FILE" | awk -v since="$SINCE_EPOCH" -v code="$CODE" '
BEGIN {
    month_map["Jan"]=1; month_map["Feb"]=2; month_map["Mar"]=3
    month_map["Apr"]=4; month_map["May"]=5; month_map["Jun"]=6
    month_map["Jul"]=7; month_map["Aug"]=8; month_map["Sep"]=9
    month_map["Oct"]=10; month_map["Nov"]=11; month_map["Dec"]=12

    since_valid = (since != 0)
    error_count = 0
}

function is_leap_year(year) {
    return (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)
}

function to_epoch(datestr,    parts, datetime, tz, dt, date_parts, d, mon, y, H, M, S, m, sign, h, min, off, timestamp, days_in_month) {
    if (datestr == "") return 0
    gsub(/^\[|\]$/, "", datestr)
    split(datestr, parts, " ")
    if (length(parts) < 2) return 0

    datetime = parts[1]; tz = parts[2]
    if (tz !~ /^[+-][0-9]{4}$/) return 0

    split(datetime, dt, ":")
    if (length(dt) < 4) return 0

    split(dt[1], date_parts, "/")
    if (length(date_parts) < 3) return 0

    d = date_parts[1] + 0; mon = date_parts[2]; y = date_parts[3] + 0
    H = dt[2] + 0; M = dt[3] + 0; S = dt[4] + 0

    m = month_map[mon]
    if (!m) return 0
    if (y < 1970 || y > 2100) return 0

    days_in_month = 31
    if (m == 4 || m == 6 || m == 9 || m == 11) days_in_month = 30
    else if (m == 2) days_in_month = is_leap_year(y) ? 29 : 28
    if (d < 1 || d > days_in_month) return 0
    if (H < 0 || H > 23 || M < 0 || M > 59 || S < 0 || S > 59) return 0

    sign = substr(tz, 1, 1)
    h = substr(tz, 2, 2) + 0
    min = substr(tz, 4, 2) + 0
    if (h > 23 || min > 59) return 0

    off = (h * 3600 + min * 60) * (sign == "+" ? 1 : -1)
    timestamp = mktime(y " " m " " d " " H " " M " " S) - off
    return (timestamp > 0 && timestamp < 2^31) ? timestamp : 0
}

{
    # ---- Разбор access.log ----
    remote_addr = $1
    remote_user = $3
    time_str = $4 " " $5
    gsub(/^\[|\]$/, "", time_str)

    n = match($0, /\"[^\"]+\"/)
    if (n > 0) {
        request = substr($0, RSTART+1, RLENGTH-2)
    } else {
        request = "-"
    }

    after = substr($0, RSTART+RLENGTH)
    split(after, p, " ")
    status = ""
    for (i=1; i<=length(p); i++) {
        if (p[i] != "") { status = p[i]; break }
    }

    if (status == "") { error_count++; next }

    # ---- Фильтрация по коду ----
    if (code == "200" && status != "200") next
    if (code == "50x" && status !~ /^5[0-9][0-9]$/) next

    # ---- Время ----
    log_epoch = to_epoch("[" time_str "]")
    if (log_epoch == 0) next

    if (!since_valid || log_epoch >= since) count++
}
END {
    if (error_count > 0) {
        print "DEBUG: skipped " error_count " invalid lines" > "/dev/stderr"
    }
    print count + 0
}' 2>>/var/log/keycloak_metrics_errors.log || echo "0")

echo "${COUNT:-0}"



++++++++++++++++++++++++++++++++++++++


1️⃣ Создание скрипта

Открываем терминал на сервере с Zabbix Agent.

Переходим в папку для пользовательских скриптов. Обычно это /usr/local/bin или /etc/zabbix/scripts (если такой нет — можно создать):

sudo mkdir -p /etc/zabbix/scripts
cd /etc/zabbix/scripts


Создаём скрипт с помощью nano:

sudo nano check_302.sh


Вставляем содержимое скрипта, например, для подсчёта строк с " 302 " в файле или архиве .gz:

#!/bin/bash

# файл передаётся как первый аргумент
FILE="$1"

if [[ "$FILE" == *.gz ]]; then
    zgrep -c " 302 " "$FILE"
else
    grep -c " 302 " "$FILE"
fi


Сохраняем и выходим из nano: Ctrl+O, Enter, Ctrl+X.

2️⃣ Установка прав и владельца

Делаем скрипт исполняемым:

sudo chmod 755 /etc/zabbix/scripts/check_302.sh


Назначаем владельца zabbix (обычно агент работает под этим пользователем):

sudo chown zabbix:zabbix /etc/zabbix/scripts/check_302.sh

3️⃣ Настройка Zabbix Agent

Открываем конфиг агента, обычно /etc/zabbix/zabbix_agentd.conf:

sudo nano /etc/zabbix/zabbix_agentd.conf


Добавляем строку UserParameter:

UserParameter=check.302[*],/etc/zabbix/scripts/check_302.sh $1


check.302[*] — ключ, который будет использоваться в Zabbix

$1 — параметр, который передастся в скрипт (например, путь к файлу или архиву)

Сохраняем файл и выходим.

4️⃣ Перезапуск агента

После изменения конфигурации необходимо перезапустить Zabbix Agent:

sudo systemctl restart zabbix-agent
sudo systemctl status zabbix-agent

5️⃣ Проверка работы

Можно проверить локально на сервере, что UserParameter работает:

zabbix_agentd -t "check.302[/var/log/nginx/access.log]"


Вывод должен показать количество строк с " 302 ".
