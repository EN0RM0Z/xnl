#!/bin/bash
#
# Отладочная версия: показывает первые 5 полей access.log
#

LOG_FILE="${LOG_FILE:-/var/log/nginx/access.log}"
MAX_LINES="${MAX_LINES:-20}"

tail -n "$MAX_LINES" "$LOG_FILE" | \
awk '
{
  # Разбиваем по кавычкам (3 части минимум)
  split($0, a, "\"")

  # 1) Первая часть: addr - user [time]
  split(a[1], p1, " ")

  remote_addr = p1[1]
  remote_user = p1[3]  # обычно "-"
  time_str   = substr(a[1], index(a[1], "[")+1, index(a[1], "]")-index(a[1], "[")-1)

  request    = a[2]

  # 2) После запроса идут: статус, размер и т.п.
  split(a[3], p2, " ")
  status = p2[2]

  # Выводим первые 5 "полей"
  print "1:", remote_addr, " 2:", remote_user, " 3:", "[" time_str "]", \
        " 4:", "\"" request "\"", " 5:", status
}
'
