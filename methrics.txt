#!/bin/bash

LOG_FILE="${LOG_FILE:-/var/log/nginx/keycloak.access.log}"
MAX_LINES="${MAX_LINES:-20}"

tail -n "$MAX_LINES" "$LOG_FILE" | \
awk '
{
    # Пример строки:
    # 192.168.1.10 - - [01/Oct/2025:07:25:03 +0300] "POST /realms/... HTTP/1.1" 200 2514

    # Первое поле — адрес
    remote_addr = $1

    # Второе поле — remote_user (может быть "-")
    remote_user = $3   # смещение из-за того, что после addr идёт "-"

    # Время в квадратных скобках: поля 4 и 5
    time_str = $4 " " $5
    gsub(/^\[|\]$/, "", time_str)

    # Вырезаем request (между кавычками)
    n = match($0, /\"[^\"]+\"/)
    if (n > 0) {
        request = substr($0, RSTART+1, RLENGTH-2)
    } else {
        request = "-"
    }

    # После кавычек идёт статус и размер
    after = substr($0, RSTART+RLENGTH)
    split(after, p, " ")
    status = ""
    for (i=1; i<=length(p); i++) {
        if (p[i] != "") {
            status = p[i]
            break
        }
    }

    # Печатаем 5 полей
    print "1:", remote_addr, \
          "2:", remote_user, \
          "3:[" time_str "]", \
          "4:\"" request "\"", \
          "5:", status
}'
